import { marked, type Tokens } from "marked"
import remend from "remend"

export type MarkdownStreamBlock = {
  key: string
  raw: string
  src: string
  mode: "full" | "live"
}

/**
 * Incremental lex cache for a streaming message. `stableBlocks` covers the
 * first `stablePrefixLength` characters of `text`; appended text can only
 * change block structure from the last meaningful token onward, so subsequent
 * ticks re-lex just the tail instead of the whole message.
 */
export type LiveMarkdownLexCache = {
  baseKey: string
  text: string
  stablePrefixLength: number
  stableBlocks: MarkdownStreamBlock[]
}

const hasReferenceDefinitions = (text: string): boolean => {
  // Up to three leading spaces are allowed (CommonMark), the space after the colon is
  // optional, and the destination may sit on the following (indented) line. Missing a
  // definition split it into a block of its own, so an earlier `[docs]` link rendered as
  // literal text until the message finalized.
  return /^[ \t]{0,3}\[[^\]]+\]:[ \t]*(?:\S|$)/m.test(text)
}

const hasOpenFence = (raw: string): boolean => {
  const match = raw.match(/^[ \t]{0,3}(`{3,}|~{3,})/)
  if (!match) return false
  const marker = match[1]
  if (!marker) return false
  const char = marker[0]
  const size = marker.length
  const last = raw.trimEnd().split("\n").at(-1)?.trim() ?? ""
  return !new RegExp(`^[\\t ]{0,3}${char}{${size},}[\\t ]*$`).test(last)
}

const healMarkdown = (text: string): string => {
  return remend(text, { linkMode: "text-only" })
}

const fnv1a32 = (input: string): string => {
  let hash = 0x811c9dc5
  for (let i = 0; i < input.length; i += 1) {
    hash ^= input.charCodeAt(i)
    hash = (hash + ((hash << 1) + (hash << 4) + (hash << 7) + (hash << 8) + (hash << 24))) >>> 0
  }
  return hash.toString(16)
}

const buildMarkdownCacheKey = (baseKey: string, raw: string, index: number, mode: "full" | "live"): string => {
  const sample = raw.length > 400 ? `${raw.slice(0, 200)}${raw.slice(-200)}` : raw
  return `${baseKey}:${index}:${mode}:${raw.length}:${fnv1a32(sample)}`
}

export const buildFullMarkdownBlock = (text: string, baseKey: string): MarkdownStreamBlock => ({
  key: buildMarkdownCacheKey(baseKey, text, 0, "full"),
  raw: text,
  src: text,
  mode: "full",
})

const buildWholeLiveBlock = (text: string, baseKey: string): MarkdownStreamBlock => ({
  key: buildMarkdownCacheKey(baseKey, text, 0, "live"),
  raw: text,
  src: healMarkdown(text),
  mode: "live",
})

const emptyCache = (baseKey: string, text: string): LiveMarkdownLexCache => ({
  baseKey,
  text,
  stablePrefixLength: 0,
  stableBlocks: [],
})

/**
 * Split streaming markdown into renderable blocks: every block except the last
 * meaningful one is final ("full"), the trailing block is "live" and re-renders
 * as it grows. When `cache` matches an append-only prefix of `text`, only the
 * tail after the cached stable prefix is re-lexed and re-healed — the output is
 * identical to lexing from scratch, but per-tick cost stays proportional to the
 * tail instead of the whole message.
 */
export const buildLiveMarkdownBlocks = (
  text: string,
  baseKey: string,
  cache?: LiveMarkdownLexCache | null,
): { blocks: MarkdownStreamBlock[]; cache: LiveMarkdownLexCache } => {
  // Reference definitions can restyle earlier blocks, so fall back to one live
  // block over the whole text.
  if (hasReferenceDefinitions(text)) {
    return { blocks: [buildWholeLiveBlock(text, baseKey)], cache: emptyCache(baseKey, text) }
  }

  const canReusePrefix =
    cache != null &&
    cache.baseKey === baseKey &&
    cache.stablePrefixLength > 0 &&
    text.length >= cache.text.length &&
    text.startsWith(cache.text)
  const stableBlocks = canReusePrefix ? cache.stableBlocks : []
  const stablePrefixLength = canReusePrefix ? cache.stablePrefixLength : 0

  let tokens: ReturnType<typeof marked.lexer>
  try {
    tokens = marked.lexer(text.slice(stablePrefixLength))
  } catch {
    // Pathologically nested or malformed input can overflow the lexer; fall back
    // to one live block so the text renders instead of crashing the chat.
    return { blocks: [buildWholeLiveBlock(text, baseKey)], cache: emptyCache(baseKey, text) }
  }

  let lastMeaningfulIndex = -1
  for (let index = 0; index < tokens.length; index += 1) {
    if (tokens[index]?.type !== "space") {
      lastMeaningfulIndex = index
    }
  }

  const blocks = stableBlocks.slice()
  let blockIndex = stableBlocks.length
  let nextStablePrefixLength = stablePrefixLength
  for (let index = 0; index < tokens.length; index += 1) {
    const token = tokens[index] as Tokens.Generic
    const raw = token.raw ?? ""
    if (token.type !== "space") {
      const mode: "full" | "live" = index === lastMeaningfulIndex ? "live" : "full"
      // A "full" block is already complete: healing it rewrote literal text in finished
      // paragraphs (`array[0` lost its bracket, `**` was appended, image-like text was
      // deleted). Only the live tail needs healing.
      const src =
        mode === "live"
          ? token.type === "code" && hasOpenFence(raw)
            ? raw
            : healMarkdown(raw)
          : raw
      blocks.push({
        key: buildMarkdownCacheKey(baseKey, raw, blockIndex, mode),
        raw,
        src,
        mode,
      })
      blockIndex += 1
    }
    if (index < lastMeaningfulIndex) {
      nextStablePrefixLength += raw.length
    }
  }

  if (blocks.length === 0) {
    return { blocks: [buildWholeLiveBlock(text, baseKey)], cache: emptyCache(baseKey, text) }
  }

  const lastBlock = blocks[blocks.length - 1]
  return {
    blocks,
    cache: {
      baseKey,
      text,
      stablePrefixLength: nextStablePrefixLength,
      stableBlocks: lastBlock?.mode === "live" ? blocks.slice(0, -1) : blocks,
    },
  }
}

import { unified } from "unified"
import remarkParse from "remark-parse"
import remarkGfm from "remark-gfm"
import remarkMath from "remark-math"
import type { Root } from "mdast"
import { buildFullMarkdownBlock, buildLiveMarkdownBlocks, type LiveMarkdownLexCache } from "./markdownStreamBlocks"
import type { MarkdownParseRequest, MarkdownParseResponse, ParsedMarkdownBlock } from "./markdownWorkerProtocol"

const parser = unified().use(remarkParse).use(remarkGfm).use(remarkMath)

const parseSyntax = (source: string): Root => parser.parse(source) as Root

export const createMarkdownWorkerParser = (parse = parseSyntax) => {
  // Retain only the last successful live request. Message changes, rewrites and
  // finalization invalidate reuse; terminating the worker releases this cache.
  let previous: { text: string; baseKey: string; lex: LiveMarkdownLexCache; blocks: ParsedMarkdownBlock[] } | null =
    null
  return (request: MarkdownParseRequest): ParsedMarkdownBlock[] => {
    const reusable = request.live && previous?.baseKey === request.baseKey && request.text.startsWith(previous.text)
    const oldBlocks = reusable ? previous!.blocks : []
    const projected = request.live
      ? buildLiveMarkdownBlocks(request.text, request.baseKey, reusable ? previous!.lex : null)
      : null
    const blocks = projected?.blocks ?? [buildFullMarkdownBlock(request.text, request.baseKey)]
    const parsed = blocks.map((block, index) => {
      const old = oldBlocks[index]
      if (
        block.mode === "full" &&
        old?.mode === "full" &&
        block.key === old.key &&
        block.raw === old.raw &&
        block.src === old.src
      )
        return old
      // Detach completed substrings from prior request buffers. V8 sliced
      // strings can otherwise retain many whole streaming snapshots. Copy UTF-16
      // units directly so unfinished surrogate pairs are preserved as well.
      const raw = block.mode === "full" ? block.raw.split("").join("") : block.raw
      const src = block.mode === "full" ? (block.src === block.raw ? raw : block.src.split("").join("")) : block.src
      return { ...block, raw, src, tree: parse(src) }
    })
    const stable = projected ? parsed.slice(0, projected.cache.stableBlocks.length) : []
    if (projected) projected.cache.stableBlocks = stable
    // The live tail is always parsed again; retaining its old AST would only
    // increase peak memory during the next parse.
    previous = projected ? { text: request.text, baseKey: request.baseKey, lex: projected.cache, blocks: stable } : null
    return parsed
  }
}

// Stateless entry point for parity checks and standalone parsing.
export const parseMarkdownInWorker = (request: MarkdownParseRequest): ParsedMarkdownBlock[] =>
  createMarkdownWorkerParser()(request)

/** Omit unchanged blocks the receiver explicitly acknowledged; send the changed suffix. */
export const createMarkdownWorkerResponder = () => {
  const parse = createMarkdownWorkerParser()
  let previous: { request: MarkdownParseRequest; blocks: ParsedMarkdownBlock[] } | null = null
  return (request: MarkdownParseRequest): MarkdownParseResponse => {
    const blocks = parse(request)
    let stablePrefixCount = 0
    if (
      request.live &&
      previous &&
      request.ownerId !== undefined &&
      request.ownerId === previous.request.ownerId &&
      request.baseKey === previous.request.baseKey &&
      request.baseRevision === previous.request.id
    ) {
      // The parser reuses references only for semantically stable blocks. A
      // reference definition, rewrite, or final parse invalidates that reuse.
      while (
        stablePrefixCount < blocks.length &&
        blocks[stablePrefixCount].mode === "full" &&
        blocks[stablePrefixCount] === previous.blocks[stablePrefixCount]
      )
        stablePrefixCount++
    }
    const response: MarkdownParseResponse = {
      id: request.id,
      blocks: stablePrefixCount ? blocks.slice(stablePrefixCount) : blocks,
      ...(stablePrefixCount ? { stablePrefixCount, baseRevision: previous!.request.id } : {}),
    }
    // One bounded response, shared with the parser, never a transcript cache.
    previous = request.live ? { request, blocks } : null
    return response
  }
}

  createMarkdownWorkerResponder,

const request = (text: string, overrides: Partial<MarkdownParseRequest> = {}): MarkdownParseRequest => ({
  id: 1,
  text,
  live: true,
  baseKey: "test",
  ...overrides,

describe("worker response transfer", () => {
  test("sends only the changed suffix against an acknowledged live revision", () => {
    const respond = createMarkdownWorkerResponder()
    const first = respond(request("# Heading\n\nStable **body**\n\nTail", { id: 1, ownerId: 1 }))
    const input = request("# Heading\n\nStable **body**\n\nTail grows", { id: 2, ownerId: 1, baseRevision: 1 })
    const next = respond(input)
    expect(next.stablePrefixCount).toBe(2)
    expect(next.blocks).toHaveLength(1)
    expect([...first.blocks!.slice(0, next.stablePrefixCount), ...next.blocks!]).toEqual(parseMarkdownInWorker(input))
  })

  test.each([
    { id: 2, ownerId: 2, baseRevision: 1 },
    { id: 2, ownerId: 1, baseRevision: 0 },
    { id: 2, ownerId: 1, baseRevision: 1, live: false },
  ])("sends a full response for owner, revision, and finalization changes", (overrides) => {
    const respond = createMarkdownWorkerResponder()
    const text = "# Heading\n\nStable body\n\nTail"
    respond(request(text, { id: 1, ownerId: 1 }))
    const input = request(text + " grows", overrides)
    const next = respond(input)
    expect(next.stablePrefixCount ?? 0).toBe(0)
    expect(next.blocks).toEqual(parseMarkdownInWorker(input))
  })

  test("a late reference definition invalidates the completed prefix", () => {
    const respond = createMarkdownWorkerResponder()
    const text = "# Heading\n\n[reference] text\n\nTail\n\n"
    respond(request(text, { id: 1, ownerId: 1 }))
    const input = request(text + "[reference]: https://example.com", { id: 2, ownerId: 1, baseRevision: 1 })
    const next = respond(input)
    expect(next.stablePrefixCount ?? 0).toBe(0)
    expect(next.blocks).toEqual(parseMarkdownInWorker(input))
  })
