import manifest from "./locales.json" with { type: "json" }
const locales = Object.keys(manifest.labels)
const isLocale = (value) => typeof value === "string" && Object.hasOwn(manifest.labels, value)

function resolveLocale(value) {
  if (typeof value !== "string") return undefined
  try {
    const tag = new Intl.Locale(value.replace(/_/g, "-"))
    const full = tag.maximize()
    const messageLocale = full.language === "zh"
      ? full.script === "Hant" ? "zh-TW" : "zh-CN"
      : full.language === "pt" ? "pt-BR" : full.language
    if (!isLocale(messageLocale)) return undefined
    return { locale: messageLocale, formatLocale: tag.toString() }
  } catch {
    return undefined
  }
}

function resolvePreferences(preferences, systemLanguages = []) {
  const selected = preferences.interface === "system"
    ? systemLanguages.map(resolveLocale).find(Boolean)
    : { locale: preferences.interface, formatLocale: preferences.interface }
  return { ...preferences, ...(selected || { locale: "en", formatLocale: "en" }) }
}

function normalizePreferences(value) {
  const valid = value && typeof value === "object" && !Array.isArray(value) && (value.schema === undefined || value.schema === 1) ? value : {}
  return {
    schema: 1,
    interface: valid.interface === "system" || isLocale(valid.interface) ? valid.interface : "system",
    conversation: valid.conversation === "auto" || isLocale(valid.conversation) ? valid.conversation : "auto",
    revision: Number.isSafeInteger(valid.revision) && valid.revision >= 0 ? valid.revision : 0,
  }
}

function validatePatch(value) {
  if (!value || typeof value !== "object" || Array.isArray(value)) throw new Error("Invalid language preferences")
  const keys = Object.keys(value)
  if (!keys.length || keys.some((key) => !["interface", "conversation"].includes(key))) {
    throw new Error("Unknown language preference")
  }
  if (Object.hasOwn(value, "interface") && value.interface !== "system" && !isLocale(value.interface)) {
    throw new Error("Unsupported interface language")
  }
  if (Object.hasOwn(value, "conversation") && value.conversation !== "auto" && !isLocale(value.conversation)) {
    throw new Error("Unsupported conversation language")
  }
  return value
}

function createLanguageController({ read, write, systemLanguages, onChange = () => {} }) {
  let preferences = normalizePreferences(read())
  let chain = Promise.resolve()
  const get = () => resolvePreferences(preferences, systemLanguages())
  return {
    get,
    update: (value) => {
      const patch = { ...validatePatch(value) }
      const next = chain.then(async () => {
        const candidate = { ...preferences, ...patch, revision: preferences.revision + 1 }
        await write(candidate)
        preferences = candidate
        const state = get()
        try { onChange(state) } catch (error) {
          console.error("Failed to notify windows of committed language preferences", error)
        }
        return state
      })
      chain = next.catch(() => {})
      return next
    },
  }
}

function conversationInstruction(locale) {
  if (!isLocale(locale)) return undefined
  return [
    `Conversation preference: write user-facing explanations and summaries in ${manifest.conversationNames[locale]}.`,
    "This is a preference only. Explicit user language requests and project instructions take precedence.",
    "Follow project rules for code comments, commits, and documents; otherwise keep these artifacts in English.",
    "Preserve technical identifiers, commands, paths, quoted evidence, original task constraints, test results, and raw tool output exactly.",
    "Do not translate evidence before aggregation or change approval scope, severity, support counts, rankings, or verification state.",
  ].join("\n")
}

export async function loadDictionary(locale: Locale): Promise<I18nDictionary> {
  const cached = dictionaries.get(locale)
  if (cached) return cached
  const pending = inFlight.get(locale)
  if (pending) return pending
  const request = loaders[locale as Exclude<Locale, "en">]().then(({ default: dictionary }) => {
    if (Object.keys(dictionary).length !== Object.keys(enDict).length ||
      Object.keys(enDict).some((key) => typeof dictionary[key as I18nKey] !== "string")) {
      throw new Error("Invalid language catalog")
    }
    dictionaries.set(locale, dictionary as I18nDictionary)
    return dictionary as I18nDictionary
  }).finally(() => inFlight.delete(locale))
  inFlight.set(locale, request)
  return request
}

async function applyPreferences(value: ResolvedLanguagePreferences): Promise<void> {
  if (value.revision < useI18nStore.getState().revision) return
  const sequence = ++requestSequence
  useI18nStore.setState({
    interface: value.interface, conversation: value.conversation, revision: value.revision,
    loadingLocale: value.locale, loadingError: false,
  })
  try {
    const dictionary = await loadDictionary(value.locale)
    if (sequence !== requestSequence) return
    useI18nStore.setState({ locale: value.locale, dictionary, formatLocale: value.formatLocale, loadingLocale: null })
  } catch {
    if (sequence !== requestSequence) return
    useI18nStore.setState({ locale: "en", dictionary: enDict, formatLocale: "en", loadingLocale: null, loadingError: true })
  }
}

function parsePreferences(value: unknown): ResolvedLanguagePreferences {
  const prefs = normalizePreferences(value)
  const state = value as Partial<ResolvedLanguagePreferences> | null
  // Main provides the preferred OS language list's resolved result. Validate
  // the formatting tag; a malformed tag must never reach Intl constructors.
  const resolved = resolvePreferences(prefs, systemLanguages())
  if (state && typeof state.locale === "string" && (Object.hasOwn(loaders, state.locale) || state.locale === "en")) {
    try {
      const formatLocale = new Intl.Locale(state!.formatLocale || state!.locale!).toString()
      return { ...prefs, locale: state!.locale as Locale, formatLocale }
    } catch { /* Use the validated local resolution below. */ }
  }
  return resolved
}

async function savePreferences(patch: Partial<Pick<LanguagePreferences, "interface" | "conversation">>): Promise<void> {
  validatePatch(patch)
  const bridge = getDesktopBridge()

    const state = await bridge.invoke("desktop_language_set", patch)

}

export function initializeLocale(): Promise<void> {
  if (initialization) return initialization
  const work = (async () => {
    const bridge = getDesktopBridge()
    if (bridge?.invoke) {
      if (bridge.listen) {
        dispose = await bridge.listen("openchamber:language-changed", ({ payload }) => {
          void applyPreferences(parsePreferences(payload))
        })
      }
      await applyPreferences(parsePreferences(await bridge.invoke("desktop_language_get")))
    } else {
      let saved: unknown
      try { saved = JSON.parse(window.localStorage.getItem(STORAGE_KEY) || "null") } catch { /* Defaults survive corrupt storage. */ }
      const listener = (event: StorageEvent) => {
        if (event.key !== STORAGE_KEY) return
        try { void applyPreferences(parsePreferences(JSON.parse(event.newValue || "null"))) } catch { /* Ignore corrupt external writes. */ }
      }
      window.addEventListener("storage", listener)
      dispose = () => window.removeEventListener("storage", listener)
      await applyPreferences(parsePreferences(saved))
    }
  })().catch(() => {
    useI18nStore.setState({ locale: "en", dictionary: enDict, formatLocale: "en", loadingLocale: null, loadingError: true })
  })
  let timer: ReturnType<typeof setTimeout> | undefined
  const timeout = new Promise<void>((resolve) => {
    timer = setTimeout(() => {
      requestSequence++
      useI18nStore.setState({ locale: "en", dictionary: enDict, formatLocale: "en", loadingLocale: null, loadingError: true })
      resolve()
    }, 3000)
  })
  initialization = Promise.race([work, timeout]).finally(() => clearTimeout(timer))
  return initialization
}

export function formatMessage(dictionary: I18nDictionary, key: I18nKey, params?: I18nParams): string {
  const template = dictionary[key] ?? enDict[key] ?? key
  if (!params) return template
  return template.replace(/\{([^{}]+)\}/g, (match, rawKey) => {
    const name = rawKey.trim()
    const value = Object.hasOwn(params, name) ? params[name] : undefined
    return value === null || value === undefined ? match : String(value)
  })
}

export function formatPluralMessage(dictionary: I18nDictionary, locale: Locale, key: "common.itemCount", count: number): string {
  const category = new Intl.PluralRules(locale).select(count)
  return formatMessage(dictionary, `${key}.${category}` as I18nKey, { count })
}

function createNativeLocalizer(getLocale, dictionaries = catalogs) {
  return (source, params) => {
    const key = Object.hasOwn(mapping, source) ? mapping[source] : undefined
    const message = key ? dictionaries[getLocale()]?.[key] ?? english[key] ?? source : source

function createLocalizedDialog(dialog, translate) {
  const options = (value) => {
    const next = { ...value }
    for (const key of ["title", "message", "detail", "buttonLabel"]) {
      if (typeof next[key] === "string") next[key] = translate(next[key])
    }
    if (Array.isArray(next.buttons)) next.buttons = next.buttons.map((button) => translate(button))
    return next
  }
  const result = {}
  for (const method of ["showMessageBox", "showOpenDialog", "showSaveDialog"]) {
    result[method] = (...args) => {
      const index = args.length - 1
      args[index] = options(args[index])
      return dialog[method](...args)
    }
  }
  result.showErrorBox = (title, content) => dialog.showErrorBox(translate(title), translate(content))
  return result
}

import ts from "typescript"

const placeholders = (value: string) =>
  [...new Set([...value.matchAll(/\{([^{}]+)\}/g)].map((match) => match[1].trim()))].sort().join("|")

export function validateCatalog(text: string, english: Record<string, string>, allowed: readonly string[] = []) {
  const errors: string[] = []
  const parsed = ts.parseJsonText("catalog.json", text)
  const seen = new Set<string>()
  const scan = (node: ts.Node) => {
    if (ts.isPropertyAssignment(node)) {
      const key = ts.isStringLiteral(node.name) ? node.name.text : node.name.getText(parsed)
      if (seen.has(key)) errors.push(`duplicate key ${key}`)
      seen.add(key)
    }
    ts.forEachChild(node, scan)
  }
  scan(parsed)
  let catalog: Record<string, string>
  try { catalog = JSON.parse(text) } catch { return { errors: [...errors, "invalid JSON"], catalog: {} } }
  if (!catalog || typeof catalog !== "object" || Array.isArray(catalog)) {
    return { errors: [...errors, "invalid catalog object"], catalog: {} }
  }
  for (const key of Object.keys(catalog)) if (!Object.hasOwn(english, key)) errors.push(`unknown key ${key}`)
  for (const [key, source] of Object.entries(english)) {
    const value = catalog[key]
    if (typeof value !== "string" || !value.trim()) errors.push(`missing or empty ${key}`)
    else {
      if (value !== value.normalize("NFC")) errors.push(`non-NFC ${key}`)
      if (placeholders(value) !== placeholders(source)) errors.push(`placeholder mismatch ${key}`)
      if (value === source && !allowed.includes(key)) errors.push(`unreviewed identical string ${key}`)
    }
  }
  return { errors, catalog }
}
