import { create } from "zustand"
import {
  captureBlockingRequest,
  type BlockingRequestDecision,
  type BlockingRequestIdentity,
} from "@/sync/session-actions"

export type RequestDecisionPhase = "idle" | "sending" | "checking" | "retry" | "uncertain" | "accepted" | "resolved"
type RecordEntry = { phase: RequestDecisionPhase; source?: ReturnType<typeof captureBlockingRequest> }

/** Ephemeral presentation state only. The runtime owns permission outcomes; this shares UI locks across cards. */
export const useRequestDecisionStore = create<{ records: ReadonlyMap<string, RecordEntry> }>(() => ({
  records: new Map(),
}))
const subscribers = new Map<string, number>()
export const requestDecisionKey = ({ kind, sessionId, requestId }: BlockingRequestIdentity) =>
  JSON.stringify([kind, sessionId, requestId])

export function retainRequestDecision(key: string) {
  subscribers.set(key, (subscribers.get(key) ?? 0) + 1)
  return () => {
    const count = (subscribers.get(key) ?? 1) - 1
    if (count) subscribers.set(key, count)
    else subscribers.delete(key)
    pruneResolvedDecisions()
  }
}

function pruneResolvedDecisions() {
  const records = new Map(useRequestDecisionStore.getState().records)
  let changed = false
  for (const [key, entry] of records) {
    if (
      !subscribers.has(key) &&
      entry.phase !== "sending" &&
      entry.phase !== "checking" &&
      entry.source?.isPendingLocally() === false
    ) {
      records.delete(key)
      changed = true
    }
  }
  if (changed) useRequestDecisionStore.setState({ records })
}

function update(key: string, entry: RecordEntry) {
  const records = new Map(useRequestDecisionStore.getState().records)
  records.set(key, entry)
  useRequestDecisionStore.setState({ records })
}

async function reconcile(identity: BlockingRequestIdentity, source?: ReturnType<typeof captureBlockingRequest>) {
  const key = requestDecisionKey(identity)
  update(key, { phase: "checking", source })
  try {
    source ??= captureBlockingRequest(identity)
    update(key, { phase: "checking", source })
    const result = await source.read()
    update(key, { phase: result === "pending" ? "retry" : "resolved", source })
  } catch {
    update(key, { phase: "uncertain", source })
  }
  pruneResolvedDecisions()
}

export async function checkRequestDecision(identity: BlockingRequestIdentity) {
  const entry = useRequestDecisionStore.getState().records.get(requestDecisionKey(identity))
  if (entry?.phase !== "uncertain") return
  await reconcile(identity, entry.source)
}

export async function submitRequestDecision(identity: BlockingRequestIdentity, decision: BlockingRequestDecision) {
  const key = requestDecisionKey(identity)
  const previous = useRequestDecisionStore.getState().records.get(key)
  if (previous && previous.phase !== "idle" && previous.phase !== "retry") return
  // Synchronous store update guards both same-tick activation and another mounted/remounted surface.
  let source = previous?.source
  update(key, { phase: "sending", source })
  try {
    source ??= captureBlockingRequest(identity)
    update(key, { phase: "sending", source })
    await source.respond(decision)
    update(key, { phase: "accepted", source })
  } catch {
    await reconcile(identity, source)
  }
  pruneResolvedDecisions()
}

import type { OpenChamberProjectAction } from "./openchamberConfig"

const hasControlCharacters = (value: string) => {
  for (const character of value) {
    const code = character.charCodeAt(0)
    if (code <= 31 || code === 127) return true
  }
  return false
}
const quotePosixArgument = (value: string) => `'${value.replaceAll("'", "'\\''")}'`

export function isSupportedRecipeShell(shell: string): boolean {
  return /(?:^|\/)(?:sh|bash|zsh)$/.test(shell)
}

/** A user-selected command recipe, never an installer or a second process supervisor. */
export function createStreamlitProjectAction(input: {
  id: string
  name: string
  interpreter: string
  entrypoint: string
  port: string
}): OpenChamberProjectAction | null {
  const interpreter = input.interpreter.trim()
  const entrypoint = input.entrypoint.trim()
  const port = Number(input.port)
  if (
    !interpreter ||
    !entrypoint ||
    [input.interpreter, input.entrypoint].some(hasControlCharacters) ||
    !/^[0-9]+$/.test(input.port) ||
    !Number.isInteger(port) ||
    port < 1 ||
    port > 65535 ||
    entrypoint.startsWith("-") ||
    !entrypoint.endsWith(".py")
  )
    return null
  return {
    id: input.id,
    name: input.name,
    command: `${quotePosixArgument(interpreter)} -m streamlit run ${quotePosixArgument(entrypoint)} --server.address=127.0.0.1 --server.port=${port} --server.headless=true`,
    icon: "server",
    autoOpenUrl: true,
    openUrl: `http://127.0.0.1:${port}/`,
    waitForUrl: true,
    shell: "posix",
  }
}

/** Streamlit prints this after binding; a command echo or an unrelated URL is not readiness. */
export function readyProjectActionUrl(output: string, expectedUrl: string): string | null {
  let expected: URL
  try {
    expected = new URL(expectedUrl)
  } catch {
    return null
  }
  if (!/^https?:$/.test(expected.protocol) || expected.username || expected.password) return null
  const escape = String.fromCharCode(27)
  const plain = output.replace(new RegExp(`${escape}\\[[0-9;?]*[ -/]*[@-~]`, "g"), "")
  for (const line of plain.split(/\r?\n/)) {
    const match = line.match(/^\s*(?:Local URL|URL):\s+(https?:\/\/\S+)\s*$/)
    if (!match) continue
    try {
      const url = new URL(match[1])
      if (!url.username && !url.password && url.origin === expected.origin) return expected.href
    } catch {
      /* Incomplete output is not ready. */
    }
  }
  return null
}

export type BlockingRequestIdentity = { kind: "permission" | "question"; sessionId: string; requestId: string }
export type BlockingRequestDecision = "once" | "reject" | string[][]

/** Capture routing before any await. Reconciliation observes this same request without auto-accepting or merging. */
export function captureBlockingRequest(identity: BlockingRequestIdentity) {
  const { kind, sessionId, requestId } = identity
  const directory =
    resolveDirectoryForBlockingRequest(kind, sessionId, requestId) || getSessionDirectory(sessionId) || dir()
  const client = getRequestReplyClient(kind, sessionId, requestId)
  const scope = directory ? { directory } : {}
  const stores = _childStores
  return {
    isPendingLocally: () => {
      if (!stores || !directory) return true
      const store = stores.children.get(directory)
      // An evicted store is not evidence that the runtime resolved the request.
      if (!store) return true
      return Boolean(store.getState()[kind][sessionId]?.some((request) => request.id === requestId))
    },
    respond: async (decision: BlockingRequestDecision): Promise<void> => {
      if (
        kind === "permission"
          ? decision !== "once" && decision !== "reject"
          : decision !== "reject" && !Array.isArray(decision)
      ) {
        throw new Error("Invalid request decision")
      }
      await waitForConnectionOrThrow()
      const result =
        kind === "permission"
          ? await client.permission.reply({ requestID: requestId, reply: decision as "once" | "reject", ...scope })
          : decision === "reject"
            ? await client.question.reject({ requestID: requestId, ...scope })
            : await client.question.reply({ requestID: requestId, answers: decision as string[][], ...scope })
      if (result.error || !result.data) throw new Error("Request response could not be confirmed")
    },
    read: async (): Promise<"pending" | "resolved"> => {
      if (!directory) throw new Error("Request scope is unavailable")
      const stamp = blockingRequestStamp(sessionId)
      const result = kind === "permission" ? await client.permission.list(scope) : await client.question.list(scope)
      if (result.error || !Array.isArray(result.data)) throw new Error("Request status is unavailable")
      if (
        result.data.some(
          (request) => !request || typeof request.id !== "string" || typeof request.sessionID !== "string",
        )
      ) {
        throw new Error("Invalid pending request response")
      }
      if (blockingRequestStamp(sessionId) !== stamp) throw new Error("Request status changed during the read")
      return result.data.some((request) => request.id === requestId && request.sessionID === sessionId)
        ? "pending"
        : "resolved"
    },
  }
}
