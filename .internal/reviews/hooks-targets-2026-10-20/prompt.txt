const formatProjectLabel = (label: string): string => {
  return label.replace(/[-_]/g, " ").replace(/\b\w/g, (char) => char.toUpperCase())
}

export const SettingsProjectSelector: React.FC<{ className?: string }> = ({ className }) => {
  const { t } = useI18n()
  const projects = useProjectsStore((state) => state.projects)
  const activeProjectId = useProjectsStore((state) => state.activeProjectId)
  const setActiveProject = useProjectsStore((state) => state.setActiveProject)

  const sortedProjects = React.useMemo(() => {
    const entries = [...projects].sort((a, b) => (a.label || a.path).localeCompare(b.label || b.path)).map(project => {
      const raw = project.label?.trim() || project.path.split("/").filter(Boolean).pop() || project.path ||
        t("settings.shared.projectSelector.fallbackProject")
      return { ...project, displayLabel: formatProjectLabel(raw).replace(/\s+/g, " ").trim() }
    })
    const counts = new Map<string, number>()
    for (const project of entries) counts.set(project.displayLabel, (counts.get(project.displayLabel) ?? 0) + 1)
    return entries.map(project => ({ ...project, ambiguous: (counts.get(project.displayLabel) ?? 0) > 1 }))
  }, [projects, t])

  const activeProject = React.useMemo(() => {
    if (sortedProjects.length === 0) {
      return null
    }
    return sortedProjects.find((p) => p.id === activeProjectId) ?? sortedProjects[0]
  }, [activeProjectId, sortedProjects])

  if (sortedProjects.length === 0) {
    return null
  }

  const label = activeProject?.displayLabel

  return (
    <div className={cn(className)}>
      <DropdownMenu>
        <DropdownMenuTrigger asChild>
          <button
            type="button"
            aria-label={t("settings.shared.projectSelector.switchProjectAria")}
            title={t("settings.shared.projectSelector.switchProjectTitle")}
            className={cn(
              // Mirror Input sizing so headers align visually.
              "text-foreground border border-border/80 appearance-none flex min-h-8 w-full min-w-0 rounded-lg bg-transparent px-3 py-1 outline-none",
              "hover:border-input focus-visible:ring-1 focus-visible:ring-primary/50 focus-visible:border-primary/70",
              "flex items-center gap-1.5 text-left",
            )}
          >
            <Icon name="folder" className="h-4 w-4 shrink-0 opacity-70" />
            <span className="min-w-0 flex-1">
              <span className="block break-all typography-ui-label font-medium">{label}</span>
              {activeProject?.ambiguous ? <span className="block break-all font-mono typography-micro text-muted-foreground">{activeProject.path}</span> : null}
            </span>
            <Icon name="arrow-down-s" className="size-4 shrink-0 opacity-50" />
          </button>
        </DropdownMenuTrigger>
        <DropdownMenuContent align="start" className="w-96 max-w-[calc(100vw-2rem)]">
          <DropdownMenuRadioGroup
            value={activeProject?.id ?? ""}
            onValueChange={(value) => {
              if (!value) return
              setActiveProject(value)
            }}
          >
            {sortedProjects.map((project) => {
              const itemLabel = project.displayLabel
              return (
                <DropdownMenuRadioItem key={project.id} value={project.id}
                  aria-label={project.ambiguous ? `${itemLabel} · ${project.path}` : itemLabel}>
                  <span className="min-w-0">
                    <span className="block break-all typography-ui-label">{itemLabel}</span>
                    {project.ambiguous ? <span className="block break-all font-mono typography-micro text-muted-foreground">{project.path}</span> : null}

  invalid: "settings.hooks.file.status.invalid",

const TRUST_ENV = "AX_CODE_TRUST_PROJECT_CONFIG=1"

const badgeClass =
  "typography-micro rounded px-1.5 py-0.5 border border-border/50 bg-[var(--surface-muted)] text-muted-foreground"

function EmphasizedTrustText({ text }: { text: string }) {
  const index = text.indexOf(TRUST_ENV)
  if (index < 0) return text
  return (
    <>
      {text.slice(0, index)}
      <code className="font-mono">{TRUST_ENV}</code>
      {text.slice(index + TRUST_ENV.length)}
    </>
  )
}

function sourceText(entry: HookEntry, t: (key: I18nKey) => string): string {
  const source = hookSourceLabel(entry)
  if (source.kind === "project") return t("settings.hooks.list.source.project")
  return source.name || t("settings.hooks.list.source.pack")
}

const HookCommand: React.FC<{ command: string; context: string }> = ({ command, context }) => {

  return (
    <div className="min-w-0 space-y-1">
      <div className="typography-micro text-muted-foreground">{t("settings.hooks.list.column.command")}</div>
      <code
        id={commandId}
        className={cn(
          "block font-mono typography-micro text-foreground",
          expanded ? "whitespace-pre-wrap break-all" : "truncate",
        )}
        title={command}
      >
        {command}
      </code>
      {canExpand ? (
        <Button
          type="button"
          size="xs"
          variant="ghost"
          aria-expanded={expanded}
          aria-controls={commandId}
          aria-label={`${expanded ? t("settings.hooks.list.command.collapse") : t("settings.hooks.list.command.expand")}: ${context}`}
          onClick={() => setExpanded((current) => !current)}
        >
          {expanded ? t("settings.hooks.list.command.collapse") : t("settings.hooks.list.command.expand")}
        </Button>
      ) : null}
    </div>
  )
}

export const HooksPage: React.FC = () => {
  const { t } = useI18n()
  const runtime = React.useContext(RuntimeAPIContext)
  // SettingsProjectSelector writes the active project here, the same directory MCP reads.
  const currentDirectory = useDirectoryStore((state) => state.currentDirectory)
  const directory = currentDirectory.trim()
  const [workspace, setWorkspace] = React.useState<HookWorkspace | null>(null)
  const [loading, setLoading] = React.useState(directory.length > 0)
  const [error, setError] = React.useState<string | null>(null)
  const requestRef = React.useRef(0)

  const load = React.useCallback((nextDirectory: string, mode: "replace" | "refresh") => {
    const requestId = ++requestRef.current
    if (mode === "replace") setWorkspace(null)
    setError(null)
    setLoading(true)
    void fetchHookWorkspace(nextDirectory).then(
      (next) => {
        if (requestId !== requestRef.current) return
        setWorkspace(next)
        setError(null)
        setLoading(false)
      },
      (reason: unknown) => {
        if (requestId !== requestRef.current) return
        if (mode === "replace") setWorkspace(null)
        setError(hooksLoadErrorMessage(reason))
        setLoading(false)
      },
    )
  }, [])

  React.useEffect(() => {
    if (!directory) {
      requestRef.current += 1
      setWorkspace(null)
      setError(null)
      setLoading(false)
      return
    }
    load(directory, "replace")
  }, [directory, load])

  const openFile = React.useCallback(
    (path: string) => {
      if (!directory || !path) return
      const editor = runtime?.editor
      if (editor) {
        void editor.openFile(path)
        return
      }
      useUIStore.getState().openContextFile(directory, path)
    },
    [directory, runtime],
  )

  const retry = () => {
    if (!directory || loading) return
    load(directory, workspace ? "refresh" : "replace")
  }

  const showSkeleton = Boolean(directory) && loading && !workspace
  const showError = error !== null && !showSkeleton

  return (
    <SettingsPageLayout>
      <div className="flex flex-wrap items-start justify-between gap-3">
        <div className="min-w-0 space-y-1">
          <h1 className="typography-ui-header font-semibold text-foreground">{t("settings.page.hooks.title")}</h1>
          <p className="typography-ui-label text-muted-foreground">{t("settings.hooks.page.description")}</p>
        </div>
        <Button
          type="button"
          size="sm"
          variant="outline"
          aria-label={t("settings.hooks.page.refreshAria")}
          disabled={!directory || loading}
          onClick={() => {
            if (!directory || loading) return
            load(directory, workspace ? "refresh" : "replace")
          }}
        >
          <Icon name="refresh" className={cn("h-4 w-4", loading && "animate-spin")} />
          {t("settings.hooks.page.refresh")}
        </Button>
      </div>

      <SettingsProjectSelector />

      {!directory ? (
        <p className="typography-ui-label text-muted-foreground">{t("settings.hooks.page.noDirectory")}</p>
      ) : null}

      {showSkeleton ? <ViewLoadingSkeleton label={t("settings.hooks.page.loading")} /> : null}

      {showError ? (
        <div role="alert" className="space-y-3 rounded-lg border border-border px-3 py-3">
          <p className="typography-ui-label text-foreground">{t("settings.hooks.page.error")}</p>
          {error ? <p className="typography-micro text-muted-foreground">{error}</p> : null}
          <Button type="button" size="sm" variant="outline" onClick={retry} disabled={loading}>
            {t("settings.hooks.page.retry")}
          </Button>
        </div>
      ) : null}

      {workspace ? <HooksWorkspaceView key={directory} workspace={workspace} onOpenFile={openFile} /> : null}
    </SettingsPageLayout>
  )
}
