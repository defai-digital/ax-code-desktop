const legacyQuote = (value: string) => `'${value.replaceAll("'", "'\\''")}'`
const quote = (value: string) => {
  const single = legacyQuote(value)
  const double = `"${value.replace(/[\\$`"]/g, (character) => `\\${character}`)}"`
  return single.length <= double.length ? single : double
}
const safeText = (value: string) =>
  !Array.from(value).some((character) => character.charCodeAt(0) < 32 || character.charCodeAt(0) === 127)

type MigratedAction = { command: string; openUrl?: string; shell?: "posix" }
export function migrateLegacyProjectAction(command: string, metadata: unknown): MigratedAction {
  if (!metadata || typeof metadata !== "object" || Array.isArray(metadata)) return { command }
  const recipe = metadata as { version?: unknown; entrypoint?: unknown; interpreter?: unknown; port?: unknown }
  const entrypoint = recipe.entrypoint
  if (
    recipe.version !== 1 ||
    typeof entrypoint !== "string" ||
    !safeText(entrypoint) ||
    !entrypoint.endsWith(".py") ||
    entrypoint.length > 1000 ||
    entrypoint.split(/[\\/]/).some((part) => !part || part === "." || part === ".." || part.startsWith("-")) ||
    /[\\:]/.test(entrypoint) ||
    (recipe.interpreter !== undefined &&
      (typeof recipe.interpreter !== "string" || !recipe.interpreter.trim() || !safeText(recipe.interpreter))) ||
    (recipe.port !== undefined &&
      (typeof recipe.port !== "number" || !Number.isInteger(recipe.port) || recipe.port < 1 || recipe.port > 65535))
  )
    return { command }
  // User-edited commands are opaque and take precedence over stale metadata.
  if (command !== `python -m streamlit run ${legacyQuote(entrypoint)}`) return { command }
  const port = recipe.port ?? 8501
  const args = `-m streamlit run ${quote(entrypoint)} --server.address=127.0.0.1 --server.headless=true --server.port=${port}`
  const migrated =

  return { command: migrated, openUrl: `http://127.0.0.1:${port}/`, shell: "posix" }

import { AxBIError } from '../shared/errors.js';
import { BaseResource } from './baseResource.js';
import type {
  DatasetItem,
  CreateDatasetInput,
  UpdateDatasetInput,
} from './types.js';

/** Bounded transfer limit; larger files can use the existing Upload Data UI. */
export const MAX_UPLOAD_BYTES = 32 * 1024 * 1024;
export interface UploadedDataset {
  database_id: number;
  dataset_id: number;
  table_name: string;
}

/** CRUD operations for datasets (tables). */
export class DatasetsResource extends BaseResource<
  DatasetItem,
  CreateDatasetInput,
  UpdateDatasetInput
> {
  protected readonly basePath = '/api/v1/dataset';
  protected readonly searchColumn = 'table_name';

  /** Upload bytes once through the existing authorized REST endpoint, outside model context. */
  async uploadFile(file: Blob, filename: string): Promise<UploadedDataset> {
    if (
      !filename ||
      /[\\/\u0000-\u001f\u007f]/.test(filename) ||
      filename.length > 255 ||
      file.size === 0 ||
      file.size > MAX_UPLOAD_BYTES
    ) {
      throw new AxBIError('Invalid upload filename or file size');
    }
    const form = new FormData();
    form.append('file', file, filename);
    const result = await this.http.request<unknown>({
      method: 'POST',
      path: '/api/v1/database/auto_upload/',
      body: form,
      retry: false,
      reauthenticate: false,
      redirect: 'error',
    });
    if (!result || typeof result !== 'object' || Array.isArray(result))
      throw new AxBIError(
        'Upload result is incomplete; inspect existing datasets before retrying',
      );
    const value = result as Record<string, unknown>;
    if (
      !Number.isSafeInteger(value.dataset_id) ||
      Number(value.dataset_id) <= 0 ||
      !Number.isSafeInteger(value.database_id) ||
      Number(value.database_id) <= 0 ||
      typeof value.table_name !== 'string' ||
      !value.table_name ||
      value.error ||
      value.success === false ||
      (value.status !== undefined &&
        !['completed', 'success', 'succeeded'].includes(
          String(value.status),
        )) ||
      (value.errors !== undefined &&
        (!Array.isArray(value.errors) || value.errors.length > 0))
    ) {
      throw new AxBIError(
        'Upload result is incomplete; inspect existing datasets before retrying',
      );
    }
    return {
      dataset_id: value.dataset_id as number,
      database_id: value.database_id as number,
      table_name: value.table_name,
    };
  }

  /** Get the columns for a dataset. */
  async getColumns(
    id: number | string,
  ): Promise<NonNullable<DatasetItem['columns']>> {
    const item = await this.getById(id);
    return item.columns ?? [];
  }

  /** Get the metrics for a dataset. */
  async getMetrics(
    id: number | string,
  ): Promise<NonNullable<DatasetItem['metrics']>> {
    const item = await this.getById(id);
    return item.metrics ?? [];
  }

  /** Refresh dataset metadata from the underlying database table. */
  async refresh(id: number | string): Promise<{ message: string }> {
    return this.http.put<{ message: string }>(`${this.basePath}/${id}/refresh`);
  }

  /** Export one or more datasets as ZIP. */
  async export(ids: number[]): Promise<Blob> {
    return this.exportZip(ids);
  }
}
